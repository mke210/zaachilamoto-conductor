// ========== REEMPLAZAR FUNCIÓN aceptarServicio ========== function aceptarServicio() { if (!currentServiceId || !servicioActual) return; // El chofer propone una tarifa const tarifaBase = servicioActual.tarifa_estimada || 40; const tarifaPropuesta = prompt( '💰 Propón la tarifa al cliente:\n\n' + 'Tarifa sugerida: $' + tarifaBase + '\n\n' + 'Ingresa el monto a cobrar:', tarifaBase ); if (!tarifaPropuesta || isNaN(tarifaPropuesta) || parseInt(tarifaPropuesta) <= 0) { showToast('⚠️ Debes ingresar una tarifa válida'); return; } const tarifaFinal = parseInt(tarifaPropuesta); database.ref(`servicios/${currentServiceId}`).update({ choferId: choferId, estado: 'tarifa_propuesta', tarifa_propuesta: tarifaFinal, choferNombre: document.getElementById('choferNameDisplay').textContent, choferPlaca: 'XYZ-123', choferRating: document.getElementById('calificacionPromedio').textContent, timestamp_propuesta: firebase.database.ServerValue.TIMESTAMP }).then(() => { document.getElementById('incomingRequest').classList.remove('active'); document.getElementById('idleState').style.display = 'block'; showToast('💰 Tarifa enviada: $' + tarifaFinal + ' - Esperando confirmación'); // Escuchar si el cliente acepta escucharConfirmacionCliente(currentServiceId); }); } // ========== AGREGAR FUNCIÓN escucharConfirmacionCliente ========== function escucharConfirmacionCliente(serviceId) { database.ref(`servicios/${serviceId}/estado`).on('value', (snap) => { const estado = snap.val(); if (estado === 'aceptado') { // Cliente aceptó - mostrar servicio activo document.getElementById('idleState').style.display = 'none'; document.getElementById('activeRide').classList.add('active'); document.getElementById('step1Address').textContent = servicioActual?.origen?.direccion || 'Cliente'; document.getElementById('step2Address').textContent = servicioActual?.destino?.direccion || 'Destino'; document.getElementById('btnEtapa').textContent = '🏍️ Iniciar Recogida'; etapaActual = 1; updateEtapasUI(); showToast('✅ Cliente aceptó - Dirígete a recogerlo'); } else if (estado === 'cancelado') { // Cliente rechazó o canceló document.getElementById('activeRide').classList.remove('active'); document.getElementById('idleState').style.display = 'block'; showToast('❌ Cliente rechazó la tarifa'); currentServiceId = null; servicioActual = null; } }); } // ========== REEMPLAZAR FUNCIÓN rechazarServicio ========== function rechazarServicio() { if (!currentServiceId) return; if (confirm('¿Rechazar este servicio?')) { database.ref(`servicios/${currentServiceId}`).update({ estado: 'rechazado', choferId: choferId, timestamp_rechazo: firebase.database.ServerValue.TIMESTAMP }).then(() => { document.getElementById('incomingRequest').classList.remove('active'); document.getElementById('idleState').style.display = 'block'; showToast('❌ Servicio rechazado'); currentServiceId = null; servicioActual = null; }); } }
